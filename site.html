📋 Como usar:

  • Clique no botão abaixo para copiar o script

local function make(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props or {}) do
        o[k] = v
    end
    if parent then
        o.Parent = parent
    end
    return o
end

local function round(o, r)
    make("UICorner", { CornerRadius = UDim.new(0, r or 4) }, o)
end

local thumbCache = {}
local function thumb(userId, ttype, size)
    local key = userId .. "|" .. tostring(ttype) .. "|" .. tostring(size)
    if thumbCache[key] then
        return thumbCache[key]
    end
    local ok, img = pcall(function()
        return (Players:GetUserThumbnailAsync(userId, ttype, size))
    end)
    if ok and img then
        thumbCache[key] = img
        return img
    end
    return ""
end

local function makeLogo(parent, size, pos)
    local holder = make("Frame", { Size = UDim2.fromOffset(size, size), Position = pos, BackgroundTransparency = 1 }, parent)

    for i = 0, 3 do
        make("Frame", {

        BackgroundColor3 = Color3.fromRGB(0, 164, 239),
        BorderSizePixel = 0,
        }, holder)
    end
    return holder
end

local function askTeleport(titulo, fn)

    tpTitle.Text = titulo

    overlay.Visible = true

    overlay.Visible = false
    pendingFn = nil
    tpBusy = false
end)
tpYes.MouseButton1Click:Connect(function()
    if tpBusy or not pendingFn then
        return
    end
    tpBusy = true
    tpStatus.Text = "Teleportando..."
    local ok, err = pcall(pendingFn)
    if not ok then
        tpStatus.Text = "Não foi possível teleportar agora. (No Studio o teleporte não funciona.)"
        warn("[Windows 10] Teleporte falhou:", err)
        tpBusy = false
    end
end)

    tpStatus.Text = "Falhou: " .. tostring(msg)

local function loadFriends(list, header)
    task.spawn(function()
        local friends = {}
        local ok, pages = pcall(function()
            return Players:GetFriendsAsync(player.UserId)
        end)
        if ok and pages then
            while #friends < 80 do
                local okp, cur = pcall(function()
                    return pages:GetCurrentPage()
                end)
                if not okp then
                    break
                end
                for _, f in ipairs(cur) do
                    table.insert(friends, f)
                end
                if pages.IsFinished then
                    break
                end
                local oka = pcall(function()
                    pages:AdvanceToNextPageAsync()
                end)
                if not oka then
                    break
                end
            end
        end
        if not list:IsDescendantOf(game) then
            return
        end
        
        local online = {}
        pcall(function()
            for _, f in ipairs(player:GetFriendsOnline(200)) do
                online[f.VisitorId] = f
            end
        end)
        
        if #friends == 0 then
            header.Text = "Nenhum amigo encontrado (ou não foi possível carregar)."
            return
        end
        header.Text = "Amigos (" .. #friends .. ")"
        
        table.sort(friends, function(a, b)
            local ao, bo = online[a.Id] ~= nil, online[b.Id] ~= nil
            if ao ~= bo then
                return ao
            end
            return (a.DisplayName or a.Username or ""):lower() < (b.DisplayName or b.Username or ""):lower()
        end)
        
        for i, f in ipairs(friends) do
            local info = online[f.Id]
            local row = make("Frame", { Size = UDim2.new(1, -30, 0, 64), BackgroundColor3 = PANEL, BorderSizePixel = 0, LayoutOrder = i }, list)
            round(row, 8)
            local av = make("ImageLabel", { Size = UDim2.fromOffset(48, 48), Position = UDim2.fromOffset(8, 8), BackgroundColor3 = Color3.fromRGB(60, 60, 64), BorderSizePixel = 0, Image = "" }, row)
            round(av, 24)
            task.spawn(function()
                local img = thumb(f.Id, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100)
                if av.Parent then
                    av.Image = img
                end
            end)
            txt(row, { Size = UDim2.new(1, -250, 0, 26), Position = UDim2.fromOffset(66, 8), Text = f.DisplayName or f.Username, TextSize = 19, Font = Enum.Font.SourceSansSemibold, TextTruncate = Enum.TextTruncate.AtEnd })
            local status = "Offline"
            local color = Color3.fromRGB(150, 150, 150)
            if info then
                status = "● " .. tostring(info.LastLocation or "Online")
                color = Color3.fromRGB(0, 200, 120)
            end
            txt(row, { Size = UDim2.new(1, -250, 0, 22), Position = UDim2.fromOffset(66, 34), Text = status, TextSize = 15, TextColor3 = color, TextTruncate = Enum.TextTruncate.AtEnd })

                        TeleportService:TeleportToPlaceInstance(info.PlaceId, info.GameId, player)
                    end)
                end)
            end
        end
    end)

local function buildPC(c)
    c.BackgroundColor3 = WHITE
    txt(c, { Size = UDim2.new(1, -40, 0, 40), Position = UDim2.fromOffset(20, 10), Text = "Sobre o seu PC", TextSize = 30, TextColor3 = LIGHT_TXT, Font = Enum.Font.SourceSansLight })
    local rows = {
    { "Nome do dispositivo", player.Name .. "-PC" },
    { "Processador", CFG.PC.CPU },
    { "RAM instalada", CFG.PC.RAM },
    { "Placa de vídeo", CFG.PC.GPU },
    { "Armazenamento", CFG.PC.Disco },
    { "Edição", CFG.PC.Windows },
    { "Tipo de sistema", "Sistema operacional de 64 bits" },
    }
    for i, r in ipairs(rows) do
        local y = 58 + (i - 1) * 32
        txt(c, { Size = UDim2.fromOffset(200, 30), Position = UDim2.fromOffset(20, y), Text = r[1], TextColor3 = Color3.fromRGB(100, 100, 100) })
        txt(c, { Size = UDim2.new(1, -250, 0, 30), Position = UDim2.fromOffset(230, y), Text = r[2], TextColor3 = LIGHT_TXT })
    end
    
    local baseY = 58 + #rows * 32 + 16
    local fpsLabel = txt(c, { Size = UDim2.new(1, -40, 0, 28), Position = UDim2.fromOffset(20, baseY), Text = "Desempenho  -  FPS: --", TextSize = 20, TextColor3 = LIGHT_TXT, Font = Enum.Font.SourceSansSemibold })
    
    local bars = {}
    local defs = {
    { "CPU", 0.10, 0.04, 0.22 },
    { "GPU", 0.14, 0.05, 0.30 },
    { "RAM", 0.32, 0.25, 0.40 },
    }
    for i, d in ipairs(defs) do
        local y = baseY + 32 + (i - 1) * 30
        txt(c, { Size = UDim2.fromOffset(60, 24), Position = UDim2.fromOffset(20, y), Text = d[1], TextColor3 = LIGHT_TXT })
        local back = make("Frame", { Size = UDim2.new(1, -200, 0, 18), Position = UDim2.fromOffset(80, y + 3), BackgroundColor3 = Color3.fromRGB(225, 225, 225), BorderSizePixel = 0 }, c)
        local fill = make("Frame", { Size = UDim2.fromScale(d[2], 1), BackgroundColor3 = ACCENT, BorderSizePixel = 0 }, back)
        local pct = txt(c, { Size = UDim2.fromOffset(60, 24), Position = UDim2.new(1, -100, 0, y), Text = "", TextColor3 = LIGHT_TXT })
        table.insert(bars, { fill = fill, pct = pct, v = d[2], min = d[3], max = d[4] })
    end
    txt(c, { Size = UDim2.new(1, -40, 0, 20), Position = UDim2.new(0, 20, 1, -26), Text = "Uso de CPU/GPU/RAM é simulado.", TextSize = 14, TextColor3 = Color3.fromRGB(130, 130, 130) })
    
    local fps, acc = 60, 0
    local conn = RunService.RenderStepped:Connect(function(dt)
        fps = fps * 0.9 + (1 / math.max(dt, 0.001)) * 0.1
        acc += dt
        if acc >= 0.5 then
            acc = 0
            fpsLabel.Text = "Desempenho  -  FPS: " .. math.floor(fps + 0.5)
            for _, b in ipairs(bars) do
                b.v = math.clamp(b.v + (math.random() - 0.5) * 0.1, b.min, b.max)
                b.fill.Size = UDim2.fromScale(b.v, 1)
                b.pct.Text = math.floor(b.v * 100) .. "%"
            end
        end
    end)
    return function()
        conn:Disconnect()
    end
end
